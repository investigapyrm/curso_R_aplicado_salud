Laboratorio R

Ejecuta estos bloques en RStudio dentro de un proyecto local.

1. Preparar paquetes

install.packages(c("tidyverse", "janitor", "broom", "gtsummary"))
library(tidyverse)
library(janitor)
library(broom)
library(gtsummary)

2. Leer datos

salud <- read.csv("../data/datasets/salud_muestra.csv")
glimpse(salud)
tabyl(salud$diagnostico)

3. Preparar variables

salud <- salud |>
  mutate(
    sexo = factor(sexo),
    diagnostico = factor(diagnostico),
    tabaquismo = factor(tabaquismo),
    actividad_fisica = factor(actividad_fisica),
    diabetes_probable = diagnostico == "Diabetes probable"
  )

4. Describir y graficar

salud |>
  group_by(diagnostico) |>
  summarise(
    n = n(),
    edad_media = mean(edad),
    imc_media = mean(imc),
    glucosa_media = mean(glucosa_mg_dl),
    .groups = "drop"
  )

ggplot(salud, aes(x = diagnostico, y = glucosa_mg_dl, fill = diagnostico)) +
  geom_boxplot(show.legend = FALSE) +
  labs(
    title = "Glucosa por diagnóstico",
    x = "Diagnóstico",
    y = "Glucosa (mg/dl)"
  ) +
  theme_minimal()

5. Modelo sencillo

modelo <- glm(
  diabetes_probable ~ edad + imc + tabaquismo,
  data = salud,
  family = binomial
)

tidy(modelo, exponentiate = TRUE, conf.int = TRUE)
